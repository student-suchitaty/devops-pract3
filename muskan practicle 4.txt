% Title
disp('Ordinary Derivatives (Without syms)');

% Define x values
x = -5:0.1:5;

% Define functions
f = x.^3 + 5*x.^2 - 7;
g = x.^3 + 7*x.^2 - 4;

% Numerical differentiation
df_dx = gradient(f, x);
dg_dx = gradient(g, x);

% Display sample output
disp('Function f = x^3 + 5x^2 - 7');
disp('Derivative of f (df/dx):');
disp(df_dx(1:10));

disp('Function g = x^3 + 7x^2 - 4');
disp('Derivative of g (dg/dx):');
disp(dg_dx(1:10));

% Plot functions and derivatives
figure;
plot(x, f, 'b', x, df_dx, 'r');
legend('f(x)', 'df/dx');
title('Function f and its Derivative');
xlabel('x');
ylabel('Value');
grid on;

figure;
plot(x, g, 'b', x, dg_dx, 'r');
legend('g(x)', 'dg/dx');
title('Function g and its Derivative');
xlabel('x');
ylabel('Value');
grid on;
